# Annual leave balance renewal — 1 January each year
#
# Rule: new_total = 18 (yearly allocation) + unused days from previous year
#        unused = max(total - used, 0)  — only accepted leave; pending ignored
#        used is reset to 0; pending requests stay reserved
#
# Also runs automatically when balances are fetched if the year has changed
# (safety net if cron is missed).
#
# Install on the VPS (crontab -e for user rh.digitify.fr):
#
#   CRON_TZ=Africa/Tunis
#   5 0 1 1 * cd /home/rh.digitify.fr/gestion_conge/backend && APP_ENV=production ./venv/bin/python manage.py renew_annual_leave_balances --scheduled >> /home/rh.digitify.fr/logs/renew-annual-leave.log 2>&1
#
# Manual run for a specific year:
#   cd ~/gestion_conge/backend
#   APP_ENV=production python manage.py renew_annual_leave_balances --year 2027
#
# Preview without writing:
#   APP_ENV=production python manage.py renew_annual_leave_balances --year 2027 --dry-run
